LANGUAGE plpgsql

SET search_path = public

-- 1. Create type for cart items matching the JSON structure from Android
-- Android sends: [{ "product_id": "uuid", "quantity": 2 }]

CREATE OR REPLACE FUNCTION place_order(
    p_cart_items JSONB,
    p_idempotency_key VARCHAR
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER -- Runs as Postgres superuser to bypass RLS for inserts
SET search_path = pg_catalog, public, pg_temp
AS $$
DECLARE
    v_user_id UUID;
    v_order_id UUID;
    v_subtotal DOUBLE PRECISION := 0;
    v_total DOUBLE PRECISION := 0;
    v_item RECORD;
    v_product RECORD;
    v_existing_order UUID;
BEGIN
    -- 1. Identify the authenticated user
    v_user_id := auth.uid();
    IF v_user_id IS NULL THEN
        RAISE EXCEPTION 'Unauthorized: User must be logged in to place an order.';
    END IF;

    -- 2. Check Idempotency (Duplicate Prevention)
    SELECT id INTO v_existing_order FROM orders WHERE idempotency_key = p_idempotency_key;
    IF v_existing_order IS NOT NULL THEN
        -- If this request was already processed, just return success with the existing order ID
        RETURN jsonb_build_object(
            'status', 'success',
            'message', 'Order already processed',
            'order_id', v_existing_order
        );
    END IF;

    -- 3. Calculate Subtotal Server-Side by looping through cart items
    -- We parse the JSONB array of cart items
    FOR v_item IN SELECT * FROM jsonb_to_recordset(p_cart_items) AS x(product_id UUID, quantity INT)
    LOOP
        -- Look up the authoritative product price
        SELECT * INTO v_product FROM products WHERE id = v_item.product_id;
        
        IF NOT FOUND THEN
            RAISE EXCEPTION 'Product with ID % does not exist.', v_item.product_id;
        END IF;

        IF v_product.availability != 'AVAILABLE' THEN
            RAISE EXCEPTION 'Product % is currently unavailable.', v_product.name;
        END IF;

        IF v_item.quantity <= 0 THEN
            RAISE EXCEPTION 'Invalid quantity for product %.', v_product.name;
        END IF;

        -- Accumulate subtotal
        v_subtotal := v_subtotal + (v_product.price * v_item.quantity);
    END LOOP;

    -- 4. Calculate final total (for now, same as subtotal, but leaves room for taxes/discounts)
    v_total := v_subtotal;

    -- 5. Insert the Order
    INSERT INTO orders (
        user_id,
        subtotal,
        total,
        order_status,
        payment_status,
        payment_method,
        idempotency_key,
        token_number
    ) VALUES (
        v_user_id,
        v_subtotal,
        v_total,
        'PLACED',
        'PAID',       -- For now assuming CASH or placeholder
        'CASH',
        p_idempotency_key,
        UPPER(SUBSTRING(uuid_generate_v4()::text FROM 1 FOR 6))
    ) RETURNING id INTO v_order_id;

    -- 6. Insert all Order Items
    FOR v_item IN SELECT * FROM jsonb_to_recordset(p_cart_items) AS x(product_id UUID, quantity INT)
    LOOP
        -- We look up the product again (cached in memory mostly)
        SELECT * INTO v_product FROM products WHERE id = v_item.product_id;

        INSERT INTO order_items (
            order_id,
            product_id,
            quantity,
            product_name,
            product_price
        ) VALUES (
            v_order_id,
            v_item.product_id,
            v_item.quantity,
            v_product.name,
            v_product.price
        );
    END LOOP;

    -- 7. Return success
    RETURN jsonb_build_object(
        'status', 'success',
        'order_id', v_order_id,
        'total', v_total
    );

EXCEPTION
    WHEN OTHERS THEN
        -- Any exception will cause PostgreSQL to automatically rollback the transaction
        RAISE EXCEPTION 'Checkout failed: %', SQLERRM;
END;
$$;

-- 4. Replace the place_order RPC to use database-level concurrency protection
CREATE OR REPLACE FUNCTION place_order(
    p_cart_items JSONB,
    p_idempotency_key VARCHAR
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public, pg_temp
AS $$
DECLARE
    v_user_id UUID;
    v_order_id UUID;
    v_subtotal DOUBLE PRECISION := 0;
    v_total DOUBLE PRECISION := 0;
    v_item RECORD;
    v_product RECORD;
    v_existing_order UUID;
BEGIN
    -- 1. Identify the authenticated user
    v_user_id := auth.uid();
    IF v_user_id IS NULL THEN
        RAISE EXCEPTION 'Unauthorized: User must be logged in to place an order.';
    END IF;

    -- 2. Fast-path Idempotency Check (saves DB cycles if request was fully processed earlier)
    SELECT id INTO v_existing_order FROM orders WHERE idempotency_key = p_idempotency_key;
    IF v_existing_order IS NOT NULL THEN
        RETURN jsonb_build_object(
            'status', 'success',
            'message', 'Order already processed',
            'order_id', v_existing_order
        );
    END IF;

    -- 3. Calculate Subtotal Server-Side by looping through cart items
    -- We parse the JSONB array of cart items
    FOR v_item IN SELECT * FROM jsonb_to_recordset(p_cart_items) AS x(product_id UUID, quantity INT)
    LOOP
        -- Look up the authoritative product price
        SELECT * INTO v_product FROM products WHERE id = v_item.product_id;
        
        IF NOT FOUND THEN
            RAISE EXCEPTION 'Product with ID % does not exist.', v_item.product_id;
        END IF;

        IF v_product.availability != 'AVAILABLE' THEN
            RAISE EXCEPTION 'Product % is currently unavailable.', v_product.name;
        END IF;

        IF v_item.quantity <= 0 THEN
            RAISE EXCEPTION 'Invalid quantity for product %.', v_product.name;
        END IF;

        -- Accumulate subtotal
        v_subtotal := v_subtotal + (v_product.price * v_item.quantity);
    END LOOP;

    -- 4. Calculate final total (for now, same as subtotal, but leaves room for taxes/discounts)
    v_total := v_subtotal;

    -- 5. Insert the Order inside an Exception Block for Race Conditions
    BEGIN
        INSERT INTO orders (
            user_id,
            subtotal,
            total,
            order_status,
            payment_status,
            payment_method,
            idempotency_key,
            token_number
        ) VALUES (
            v_user_id,
            v_subtotal,
            v_total,
            'PLACED',
            'PAID',
            'CASH',
            p_idempotency_key,
            UPPER(SUBSTRING(uuid_generate_v4()::text FROM 1 FOR 6))
        ) RETURNING id INTO v_order_id;
    EXCEPTION WHEN unique_violation THEN
        -- Race condition: another concurrent transaction just inserted this key
        SELECT id INTO v_existing_order FROM orders WHERE idempotency_key = p_idempotency_key;
        RETURN jsonb_build_object(
            'status', 'success',
            'message', 'Order already processed (concurrent)',
            'order_id', v_existing_order
        );
    END;

    -- 6. Insert all Order Items
    FOR v_item IN SELECT * FROM jsonb_to_recordset(p_cart_items) AS x(product_id UUID, quantity INT)
    LOOP
        -- We look up the product again (cached in memory mostly)
        SELECT * INTO v_product FROM products WHERE id = v_item.product_id;

        INSERT INTO order_items (
            order_id,
            product_id,
            quantity,
            product_name,
            product_price
        ) VALUES (
            v_order_id,
            v_item.product_id,
            v_item.quantity,
            v_product.name,
            v_product.price
        );
    END LOOP;

    -- 7. Return success
    RETURN jsonb_build_object(
        'status', 'success',
        'order_id', v_order_id,
        'total', v_total
    );

EXCEPTION
    WHEN OTHERS THEN
        RAISE EXCEPTION 'Checkout failed: %', SQLERRM;
END;
$$;

-- 2. Create a robust BEFORE UPDATE trigger function to protect security columns.
-- This guarantees that even if an RLS policy permits an UPDATE to a row,
-- unauthorized roles absolutely cannot manipulate critical fields.
CREATE OR REPLACE FUNCTION prevent_student_privilege_escalation()
RETURNS trigger AS $$
BEGIN
    -- Evaluate if the authenticated user performing the action is an ADMIN.
    -- We use a direct SELECT here because auth.uid() is the session token.
    IF NOT EXISTS (SELECT 1 FROM public.users WHERE id = auth.uid() AND role = 'ADMIN') THEN
        
        -- If NOT an admin (e.g., a STUDENT), forcefully revert critical fields to their existing OLD values.
        -- This silently ignores malicious JSON payloads like {"role": "ADMIN"}.
        NEW.id := OLD.id;
        NEW.role := OLD.role;
        NEW.created_at := OLD.created_at;
        
        -- Students are only allowed to modify non-critical fields (full_name, phone, department, etc.)
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 3. Attach the trigger to the users table.
-- It fires before any update, acting as a final column-level security gate.
DROP TRIGGER IF EXISTS enforce_role_protection ON users;
CREATE TRIGGER enforce_role_protection
    BEFORE UPDATE ON users
    FOR EACH ROW
    EXECUTE FUNCTION prevent_student_privilege_escalation();

-- PHASE 14: P0 PRIVILEGE FIX - HARDENED
-- ==========================================

-- 1. Replace the trigger function with hardened security and search_path
CREATE OR REPLACE FUNCTION public.prevent_student_privilege_escalation()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public, pg_temp
AS $$
DECLARE
    v_calling_role TEXT;
    v_is_admin BOOLEAN := false;
BEGIN
    -- 1. Identify the current execution context role
    -- PostgREST explicitly sets 'role' to 'authenticated', 'anon', or 'service_role'.
    -- Direct Postgres DB connections will return their login role (e.g., 'postgres').
    v_calling_role := current_setting('role', true);
    
    -- 2. Trusted internal superusers and service backends strictly bypass this protection
    IF v_calling_role IN ('service_role', 'postgres', 'supabase_admin') THEN
        RETURN NEW;
    END IF;

    -- 3. If an authenticated REST user is making the request, verify if they are genuinely an ADMIN
    IF v_calling_role = 'authenticated' AND auth.uid() IS NOT NULL THEN
        SELECT EXISTS (
            SELECT 1 FROM public.users 
            WHERE id = auth.uid() AND role = 'ADMIN'
        ) INTO v_is_admin;
    END IF;
    
    -- 4. If NOT an Admin, explicitly reject attempts to escalate or tamper with protected fields
    IF NOT v_is_admin THEN
        IF NEW.role IS DISTINCT FROM OLD.role THEN
            RAISE EXCEPTION 'Unauthorized modification of protected user field: role';
        END IF;
        
        IF NEW.id IS DISTINCT FROM OLD.id THEN
            RAISE EXCEPTION 'Unauthorized modification of protected user field: id';
        END IF;

        IF NEW.created_at IS DISTINCT FROM OLD.created_at THEN
            RAISE EXCEPTION 'Unauthorized modification of protected user field: created_at';
        END IF;
    END IF;
    
    RETURN NEW;
END;
$$;

-- 2. Explicitly set the function owner to postgres (Locks down the SECURITY DEFINER context)
ALTER FUNCTION public.prevent_student_privilege_escalation() OWNER TO postgres;

-- 3. (Optional but safe) Reattach the trigger to ensure it's active
DROP TRIGGER IF EXISTS enforce_role_protection ON public.users;
CREATE TRIGGER enforce_role_protection
    BEFORE UPDATE ON public.users
    FOR EACH ROW

-- 2. Create or Replace place_order with the fix
CREATE OR REPLACE FUNCTION place_order(
    p_cart_items JSONB,
    p_idempotency_key VARCHAR
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public, pg_temp
AS $$
DECLARE
    v_user_id UUID;
    v_order_id UUID;
    v_subtotal DOUBLE PRECISION := 0;
    v_total DOUBLE PRECISION := 0;
    v_item RECORD;
    v_product RECORD;
    v_existing_order UUID;
    
    v_today DATE;
    v_next_token INTEGER;
    v_token_string VARCHAR(50);
BEGIN
    -- 1. Identify the authenticated user
    v_user_id := auth.uid();
    IF v_user_id IS NULL THEN
        RAISE EXCEPTION 'Unauthorized: User must be logged in to place an order.';
    END IF;

    -- 2. Fast-path Idempotency Check (saves DB cycles if request was fully processed earlier)
    SELECT id INTO v_existing_order FROM orders WHERE idempotency_key = p_idempotency_key;
    IF v_existing_order IS NOT NULL THEN
        RETURN jsonb_build_object(
            'status', 'success',
            'message', 'Order already processed',
            'order_id', v_existing_order
        );
    END IF;

    -- 3. Calculate Subtotal Server-Side by looping through cart items
    FOR v_item IN SELECT * FROM jsonb_to_recordset(p_cart_items) AS x(product_id UUID, quantity INT)
    LOOP
        SELECT * INTO v_product FROM products WHERE id = v_item.product_id;
        
        IF NOT FOUND THEN
            RAISE EXCEPTION 'Product with ID % does not exist.', v_item.product_id;
        END IF;

        IF v_product.availability != 'AVAILABLE' THEN
            RAISE EXCEPTION 'Product % is currently unavailable.', v_product.name;
        END IF;

        IF v_item.quantity <= 0 THEN
            RAISE EXCEPTION 'Invalid quantity for product %.', v_product.name;
        END IF;

        v_subtotal := v_subtotal + (v_product.price * v_item.quantity);
    END LOOP;

    v_total := v_subtotal;

    -- 4. Generate Daily Token (Concurrency Safe & Asia/Kolkata Timezone)
    -- We extract the date based on India standard time to handle rollover properly
    v_today := (CURRENT_TIMESTAMP AT TIME ZONE 'Asia/Kolkata')::DATE;
    
    INSERT INTO public.daily_token_counter (date_key, last_token)
    VALUES (v_today, 1)
    ON CONFLICT (date_key) 
    DO UPDATE SET last_token = daily_token_counter.last_token + 1
    RETURNING last_token INTO v_next_token;

    -- Format token, e.g., 'A041'
    v_token_string := 'A' || LPAD(v_next_token::TEXT, 3, '0');

    -- 5. Insert the Order inside an Exception Block for Race Conditions
    BEGIN
        INSERT INTO orders (
            user_id,
            subtotal,
            total,
            order_status,
            payment_status,
            payment_method,
            idempotency_key,
            token_number
        ) VALUES (
            v_user_id,
            v_subtotal,
            v_total,
            'PLACED',
            'PAID',
            'ONLINE',
            p_idempotency_key,
            v_token_string
        ) RETURNING id INTO v_order_id;
    EXCEPTION WHEN unique_violation THEN
        -- Race condition: another concurrent transaction just inserted this key
        SELECT id INTO v_existing_order FROM orders WHERE idempotency_key = p_idempotency_key;
        
        -- Safe fallback: we return the existing order
        RETURN jsonb_build_object(
            'status', 'success',
            'message', 'Order already processed (concurrent)',
            'order_id', v_existing_order
        );
    END;

    -- 6. Insert all Order Items
    FOR v_item IN SELECT * FROM jsonb_to_recordset(p_cart_items) AS x(product_id UUID, quantity INT)
    LOOP
        SELECT * INTO v_product FROM products WHERE id = v_item.product_id;

        INSERT INTO order_items (
            order_id,
            product_id,
            quantity,
            product_name,
            product_price
        ) VALUES (
            v_order_id,
            v_item.product_id,
            v_item.quantity,
            v_product.name,
            v_product.price
        );
    END LOOP;

    -- 7. Return success
    RETURN jsonb_build_object(
        'status', 'success',
        'order_id', v_order_id,
        'total', v_total
    );

EXCEPTION
    WHEN OTHERS THEN
        RAISE EXCEPTION 'Checkout failed: %', SQLERRM;
END;
$$;
